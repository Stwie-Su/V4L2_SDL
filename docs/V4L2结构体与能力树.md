# V4L2 结构体速查（按流程走）

> 图例：🔍 只在启动前问一遍（问完即弃）｜⚙️ 进入采集流程，每帧都用
>
> 一句话总纲：**探测是问"选择题的选项"，配置是交卷，采集是循环批改。**

---

## 一、流程：结构体挂在哪几步

### 探测流程 🔍（`v4l2_probe.c`，只跑一次）

```
QUERYCAP              ENUM_FMT           ENUM_FRAMESIZES     ENUM_FRAMEINTERVALS
你是谁/能干什么  →   有哪些像素格式  →   每个格式哪些尺寸  →   每个尺寸哪些帧率
                                                                ↓
                                                           G_PARM（当前帧率，可选）
```

特征：全部用 `index` 当**游标**循环；ioctl 返回 -1（EINVAL）就是枚举完了，**不是错误**。
这些结构体问完就不用，不进数据流。

### 采集流程 ⚙️（`v4l2_capture.c`，九步 + 收尾）

```
配置（一次）                    每帧循环                     收尾
─────────────────              ────────────────             ────────
S_FMT        定格式             DQBUF   取帧                 STREAMOFF
REQBUFS      要缓冲               ↓ 解码显示（用 bytesused）  munmap
  └ QUERYBUF+mmap×N               ↓                        close
  └ QBUF×N     全部入队          QBUF    还缓冲
STREAMON      开流
```

### 每步挂什么结构体

| 流程步 | ioctl | 结构体 | 关键字段 |
|---|---|---|---|
| 探测·门禁 | QUERYCAP | `v4l2_capability` 🔍 | `device_caps` |
| 探测·格式 | ENUM_FMT | `v4l2_fmtdesc` 🔍 | `index`（入）→ `pixelformat`（出） |
| 探测·尺寸 | ENUM_FRAMESIZES | `v4l2_frmsizeenum` 🔍 | `pixel_format`（入）→ `discrete.width/height`（出） |
| 探测·帧率 | ENUM_FRAMEINTERVALS | `v4l2_frmivalenum` 🔍 | `pixel_format+width+height`（入）→ `discrete`（出） |
| 配置·格式 | S_FMT | `v4l2_format` ⚙️ | `type` + `fmt.pix` |
| 配置·缓冲 | REQBUFS | `v4l2_requestbuffers` ⚙️ | `count / type / memory` |
| 每帧·钥匙 | QUERYBUF / QBUF / DQBUF | `v4l2_buffer` ⚙️ | `index / bytesused / m.offset` |

## 二、按流程讲结构体

### 1. `v4l2_capability` —— 门禁 🔍（QUERYCAP）

```c
struct v4l2_capability {
	__u8	driver[16];      /* "uvcvideo" */
	__u8	card[32];        /* 设备名 */
	__u32	capabilities;    /* 整个设备的能力（多节点相同） */
	__u32	device_caps;     /* ★ 本节点的能力 */
};
```

唯一要记住的判断写法：

```c
uint32_t caps = (cap.capabilities & V4L2_CAP_DEVICE_CAPS)
                ? cap.device_caps : cap.capabilities;
if (!(caps & V4L2_CAP_VIDEO_CAPTURE) || !(caps & V4L2_CAP_STREAMING))
    return -1;                       /* 不可流式采集，直接退 */
```

**为什么不用 `capabilities`**：一个 USB 摄像头占两个节点（`/dev/video0` 图像 + `/dev/video1` 元数据），`capabilities` 是合并全集、两者相同；只有 `device_caps` 分得清谁是采集节点。

本机实测：`video0 → 0x04200001`（CAPTURE\|STREAMING ✅），`video1 → 0x04a00000`（META ❌）。

---

### 2. 探测三兄弟 🔍（三层嵌套，后层拿前层输出当输入）

```
ENUM_FMT ──→ pixelformat（fourcc）
                 ↓ 作为输入
ENUM_FRAMESIZES ──→ width / height
                 ↓ 作为输入
ENUM_FRAMEINTERVALS ──→ v4l2_fract
```

**`v4l2_fmtdesc`（有哪些格式）**：输入 `index`+`type`，输出 `pixelformat`（FourCC，程序判断格式的唯一依据）。

```c
struct v4l2_fmtdesc fd = { .type = V4L2_BUF_TYPE_VIDEO_CAPTURE };
for (fd.index = 0; ioctl(dev_fd, VIDIOC_ENUM_FMT, &fd) == 0; fd.index++)
    /* fd.pixelformat / fd.description[32] */
```

**`v4l2_frmsizeenum`（某格式有哪些尺寸）**：输入 `pixel_format`，输出 `discrete.width/height`。
`type` 返回结构形态（摄像头基本都是 `DISCRETE` 离散档位，非离散就跳过 `continue`）。

**`v4l2_frmivalenum`（某尺寸有哪些帧率）**：输入 `pixel_format`+`width`+`height` 三个，输出 `v4l2_fract`：

```c
printf("%.1f fps", (double)iv.discrete.denominator / iv.discrete.numerator);
```

> 帧率 = denominator / numerator（取分数不用浮点，`1/30` 能精确表示）。
> **帧率是「格式 × 宽 × 高」三维属性**——所以 1080p MJPG 能 30fps、切 YUYV 只剩 5fps。

---

### 3. `v4l2_format` —— 配置交卷 ⚙️（S_FMT）

```c
struct v4l2_format {
	__u32	type;                /* ★ 钥匙：VIDEO_CAPTURE → 用 fmt.pix */
	union {
		struct v4l2_pix_format	pix;   /* ← 本项目只碰这个 */
		struct v4l2_meta_format	meta;  /* 元数据节点才用 */
		/* window / vbi / sdr / raw_data[200] ... 都不用管 */
	} fmt;
};
```

`type` 必须先设，union 成员才不会被误读。

**真正要填的只有 `v4l2_pix_format` 里四个字段**：

| 字段 | 干什么 |
|---|---|
| `width` / `height` | 我要的分辨率 |
| `pixelformat` | 我要的格式，如 `V4L2_PIX_FMT_MJPEG` |
| `field` | 填 `V4L2_FIELD_ANY` 即可 |
| `sizeimage` | **驱动算出来的**，缓冲上界（MJPG 下别拿来当帧长） |

**「交卷」流程——输入期望，输出实际，必须回读**：

```c
struct v4l2_format fmt = {0};
fmt.type                = V4L2_BUF_TYPE_VIDEO_CAPTURE;
fmt.fmt.pix.width       = 640;          /* 期望值 */
fmt.fmt.pix.height      = 480;
fmt.fmt.pix.pixelformat = V4L2_PIX_FMT_MJPEG;
fmt.fmt.pix.field       = V4L2_FIELD_ANY;
ioctl(fd, VIDIOC_S_FMT, &fmt);          /* 返回后 fmt 已被驱动改写成实际值 */

/* ★ 回读校验：驱动可能"讲价" */
if (fmt.fmt.pix.pixelformat != V4L2_PIX_FMT_MJPEG) {
    fprintf(stderr, "驱动不支持 MJPG，实际给了 0x%08x\n", fmt.fmt.pix.pixelformat);
    /* 不校验的后果：YUYV 数据被当 JPEG 解 → 花屏且查不到原因 */
}
```

本机实测 640×480 MJPG：`bytesperline = 0`（压缩格式正常），`sizeimage = 184320`。
MJPG 缓冲里装的是**整段 JPEG 字节流**（`FF D8 ... FF D9`），不是像素，所以 `stbi_load_from_memory()` 能直接解。

### 4. `v4l2_requestbuffers` —— 要缓冲 ⚙️（REQBUFS）

```c
struct v4l2_requestbuffers {
	__u32	count;          /* 项目要 4 个；驱动可能改小 */
	__u32	type;           /* VIDEO_CAPTURE，与其他处一致 */
	__u32	memory;         /* ★ V4L2_MEMORY_MMAP */
};
```

```c
struct v4l2_requestbuffers req = {0};
req.count = 4;  req.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;  req.memory = V4L2_MEMORY_MMAP;
ioctl(fd, VIDIOC_REQBUFS, &req);
if (req.count < 2) return -1;            /* count 可能被改小：查够不够 */
```

- `memory = MMAP`：驱动分配、mmap 映射到用户态，**零拷贝**（本项目）。
- **count 至少 2**：驱动填一块时你得处理另一块；只有 1 块必然丢帧。

---

### 5. `v4l2_buffer` —— 缓冲的"生命周期钥匙" ⚙️

一个结构体管三件事：**缓冲在哪**（`m.offset`）、**这个帧多长**（`bytesused`）、**这帧的状态**（`timestamp/sequence/flags`）。

```c
struct v4l2_buffer {
	__u32	index;          /* 0..count-1，操作哪块缓冲 */
	__u32	type;           /* VIDEO_CAPTURE */
	__u32	memory;         /* MMAP */
	__u32	bytesused;      /* ★ 这帧实际字节：唯一可信的帧长 */
	__u32	flags;          /* V4L2_BUF_FLAG_ERROR → 丢弃 */
	__u32	sequence;       /* 帧号，单调递增，断号=丢帧 */
	struct timeval	timestamp;  /* 驱动打的采集时刻 */
	union m { __u32 offset; ... };      /* ★ MMAP 下给 mmap() 用 */
	__u32	length;         /* 缓冲长度 = sizeimage */
};
```

**生命周期（与九步对应）**：

```c
/* 初始化：每块缓冲问一次"在哪"，然后 mmap */
buf.index = i;  buf.type = ...;  buf.memory = V4L2_MEMORY_MMAP;
ioctl(fd, VIDIOC_QUERYBUF, &buf);                          /* → m.offset, length */
bufs[i] = mmap(NULL, buf.length, PROT_READ|PROT_WRITE, MAP_SHARED, fd, buf.m.offset);
ioctl(fd, VIDIOC_QBUF, &buf);                              /* 空缓冲入队 */

/* 每帧 */
ioctl(fd, VIDIOC_DQBUF, &out);          /* 阻塞取一帧 → out.index 填好了 */
process(bufs[out.index], out.bytesused); /* ★ 长度用 bytesused */
ioctl(fd, VIDIOC_QBUF, &out);            /* 处理完还回去 */
```

**三个联动**（比字段列表更重要）：

```
sizeimage  →  buffer.length  →  mmap 的长度
buffer.m.offset  →  mmap() 的最后一个参数
buffer.bytesused ──→  解码/拷贝用的长度
```

**`sequence` 的实战价值**：序号连续 + 时间戳间隔稳定 = 硬件就这么慢（本机 864×480 ≈ 5.8 fps 是带宽瓶颈）；序号断号 = 程序跟不上、丢了帧。

---

### 6. `v4l2_streamparm` —— 读当前帧率 🔍（G_PARM，可选）

用 `VIDIOC_G_PARM` 读**当前生效**的帧率：`parm.capture.timeperframe`（一个 `v4l2_fract`）。只在 probe 里打一行，不参与采集。

### 其他结构体（一句话都可能嫌多）

`v4l2_window`（overlay）、`v4l2_vbi_format`（电视）、`v4l2_sdr_format`（收音机）、`v4l2_pix_format_mplane`（多平面）、`v4l2_timecode`、`v4l2_edid`——本项目全不用。看到它们别慌：那是在做别的设备/场景的事。

## 三、记 4 条规则就够了

1. **`type` 四处一致**：format / requestbuffers / buffer 里都写 `V4L2_BUF_TYPE_VIDEO_CAPTURE`，不一致驱动直接 `EINVAL`；它还决定 union 里哪个成员有效（`fmt.type` 选错 → 读出垃圾）。
2. **ioctl 是协商不是命令**：`S_FMT`、`REQBUFS` 的输入是期望值，返回后结构体已被改写成实际值——**必回读校验**，别假设驱动听话。
3. **`bytesused` 是帧长真值**：`sizeimage`/`length` 只是缓冲上界。MJPG 变长，用上界当帧长会读到上一帧残留 → 花屏。
4. **ENUM 靠 `index` 游标**：从 0 递增，返回 -1（EINVAL）即枚举完，是正常结束不是错误。

## 四、一页速查（两种流程各一段代码）

```c
/* ========== 探测 🔍 ========== */
struct v4l2_capability cap;
ioctl(fd, VIDIOC_QUERYCAP, &cap);
uint32_t caps = (cap.capabilities & V4L2_CAP_DEVICE_CAPS)
                ? cap.device_caps : cap.capabilities;
if (!(caps & V4L2_CAP_VIDEO_CAPTURE) || !(caps & V4L2_CAP_STREAMING))
    return -1;                                     /* 门禁 */

struct v4l2_fmtdesc fmtdesc = {0};
fmtdesc.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
for (fmtdesc.index = 0; ioctl(fd, VIDIOC_ENUM_FMT, &fmtdesc) == 0; fmtdesc.index++)
    ; /* fmtdesc.pixelformat（fourcc） */

struct v4l2_frmsizeenum size = {0};
size.pixel_format = fmtdesc.pixelformat;                 /* ← 上一层输出当输入 */
for (size.index = 0; ioctl(fd, VIDIOC_ENUM_FRAMESIZES, &size) == 0; size.index++)
    ; /* size.discrete.width / height */

struct v4l2_frmivalenum iv = {0};
iv.pixel_format = fmtdesc.pixelformat;
iv.width = size.discrete.width;  iv.height = size.discrete.height;
for (iv.index = 0; ioctl(fd, VIDIOC_ENUM_FRAMEINTERVALS, &iv) == 0; iv.index++)
    ; /* fps = iv.discrete.denominator / iv.discrete.numerator */

/* ========== 采集 ⚙️ ========== */
struct v4l2_format fmt = {0};                            /* 交卷 + 回读 */
fmt.type = V4L2_BUF_TYPE_VIDEO_CAPTURE;
fmt.fmt.pix.width = W;  fmt.fmt.pix.height = H;
fmt.fmt.pix.pixelformat = V4L2_PIX_FMT_MJPEG;
fmt.fmt.pix.field = V4L2_FIELD_ANY;
ioctl(fd, VIDIOC_S_FMT, &fmt);
/* 校验 fmt.fmt.pix.{pixelformat,width,height} == 期望 */

struct v4l2_requestbuffers req = {0};
req.count = 4; req.type = V4L2_BUF_TYPE_VIDEO_CAPTURE; req.memory = V4L2_MEMORY_MMAP;
ioctl(fd, VIDIOC_REQBUFS, &req);

for (i = 0; i < req.count; i++) {
    struct v4l2_buffer buf = {0};
    buf.type = V4L2_BUF_TYPE_VIDEO_CAPTURE; buf.memory = V4L2_MEMORY_MMAP; buf.index = i;
    ioctl(fd, VIDIOC_QUERYBUF, &buf);
    bufs[i] = mmap(NULL, buf.length, PROT_READ|PROT_WRITE, MAP_SHARED, fd, buf.m.offset);
    ioctl(fd, VIDIOC_QBUF, &buf);
}
ioctl(fd, VIDIOC_STREAMON, &type);

while (running) {                                        /* 每帧循环 */
    struct v4l2_buffer out = {0};
    out.type = V4L2_BUF_TYPE_VIDEO_CAPTURE; out.memory = V4L2_MEMORY_MMAP;
    ioctl(fd, VIDIOC_DQBUF, &out);
    /* ★ 长度用 out.bytesused，不是 sizeimage；out.index 定位缓冲 */
    process(bufs[out.index], out.bytesused);
    ioctl(fd, VIDIOC_QBUF, &out);
}

ioctl(fd, VIDIOC_STREAMOFF, &type);                      /* 收尾 */
```
